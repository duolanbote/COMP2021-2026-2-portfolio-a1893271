using System.Collections.Generic;
using System.Linq;

namespace AiDijkstra;

/// <summary>
/// Undirected weighted graph with Dijkstra shortest path support.
/// </summary>
public sealed class WeightedGraph
{
    private readonly Dictionary<string, List<(string Target, int Weight)>> _adjacency = new();

    public IReadOnlyCollection<string> Nodes => _adjacency.Keys;

    public void AddNode(string name)
    {
        if (!_adjacency.ContainsKey(name))
        {
            _adjacency[name] = new List<(string, int)>();
        }
    }

    public void AddUndirectedEdge(string from, string to, int weight)
    {
        AddNode(from);
        AddNode(to);

        _adjacency[from].Add((to, weight));
        _adjacency[to].Add((from, weight));
    }

    /// <summary>
    /// Runs Dijkstra from <paramref name="source"/> and returns both the total
    /// distance to every reachable node and the predecessors needed to rebuild a path.
    /// </summary>
    public (Dictionary<string, int> Distance, Dictionary<string, string?> Previous) RunDijkstra(string source)
    {
        var distance = _adjacency.Keys.ToDictionary(node => node, _ => int.MaxValue);
        var previous = _adjacency.Keys.ToDictionary(node => node, _ => (string?)null);

        distance[source] = 0;

        var frontier = new PriorityQueue<string, int>();
        frontier.Enqueue(source, 0);

        while (frontier.TryDequeue(out var current, out var priority))
        {
            // stale entry: a shorter route to this node was already processed
            if (priority > distance[current])
            {
                continue;
            }

            foreach (var (target, weight) in _adjacency[current])
            {
                var candidate = distance[current] + weight;

                if (candidate >= distance[target])
                {
                    continue;
                }

                distance[target] = candidate;
                previous[target] = current;
                frontier.Enqueue(target, candidate);
            }
        }

        return (distance, previous);
    }

    public List<string> FindShortestPath(string source, string target, out int totalDistance)
    {
        var (distance, previous) = RunDijkstra(source);

        if (distance[target] == int.MaxValue)
        {
            totalDistance = -1;
            return new List<string>();
        }

        totalDistance = distance[target];

        var path = new List<string> { target };
        var cursor = previous[target];

        while (cursor is not null)
        {
            path.Add(cursor);
            cursor = previous[cursor];
        }

        path.Reverse();
        return path;
    }
}
